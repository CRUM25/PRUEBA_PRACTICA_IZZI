def plot_hist(data, column_name, bins=30, color="skyblue", edgecolor="black"):
    for col in column_name:
        plt.hist(data[col], bins=bins, color=color, edgecolor=edgecolor)
        plt.title(f"Histograma {col}")
        plt.xlabel("Valores")
        plt.ylabel("Frecuencia")
        plt.show()
    return


def plotbox(data, column_name, cluster_column):
    plt.figure(figsize=(15, 12))
    for i, v in enumerate(column_name, 1):
        plt.subplot(2, 2, i)
        sns.boxplot(
            x=cluster_column, y=v, data=data, hue=cluster_column, palette="Set2"
        )
        plt.title(f"Distribución de {v}")
        plt.xlabel("Clúster")
        plt.ylabel("Valores")

    plt.tight_layout()
    plt.show()


def plotbarra_acum(data, col_cluster, list_cat):

    for col_cat in list_cat:
        conteo_absoluto = pd.crosstab(data[col_cluster], data[col_cat])
        proporciones = (
            pd.crosstab(data[col_cluster], data[col_cat], normalize="index") * 100
        )

        totales_por_cluster = conteo_absoluto.sum(axis=1)
        etiquetas_x = [
            f"Cluster {cluster}\n(N = {totales_por_cluster[cluster]:,})"
            for cluster in proporciones.index
        ]

        fig, ax = plt.subplots(figsize=(12, 7))
        proporciones.plot(
            kind="bar", stacked=True, ax=ax, colormap="Spectral", width=0.75
        )

        ax.set_title(
            f'Composición Porcentual de "{col_cat}" por Cluster',
            fontsize=14,
            fontweight="bold",
            pad=20,
        )
        ax.set_xlabel("Cluster (Tamaño de Muestra N)", fontsize=11, labelpad=10)
        ax.set_ylabel("Porcentaje", fontsize=11)
        ax.set_xticklabels(etiquetas_x, rotation=0, fontsize=10)
        ax.set_ylim(0, 100)

        ax.legend(
            title=col_cat, bbox_to_anchor=(1.02, 1), loc="upper left", frameon=True
        )
        ax.set_axisbelow(True)

        plt.tight_layout()
        plt.show()


def prueb_hipotesis(df, col_cluster, cols_continuas=None, cols_categoricas=None):

    resultados = []

    # PRUEBA KRUSKAL-WALLIS (VARIABLES CONTINUAS)
    for col in cols_continuas:
        df_clean = df.dropna(subset=[col_cluster, col])
        grupos = [grupo[col].values for _, grupo in df_clean.groupby(col_cluster)]
        stat, p_val = stats.kruskal(*grupos)
        resultados.append(
            {
                "Variable": col,
                "Prueba Estadística": "Kruskal-Wallis (H)",
                "Estadístico": round(stat, 4),
                "p-value (numérico)": p_val,
            }
        )

    # CHI-CUADRADO (VARIABLES CATEGÓRICAS)
    for col in cols_categoricas:
        df_clean = df.dropna(subset=[col_cluster, col])
        tabla_contingencia = pd.crosstab(df_clean[col_cluster], df_clean[col])
        chi2, p_val, dof, ex = stats.chi2_contingency(tabla_contingencia)
        resultados.append(
            {
                "Variable": col,
                "Prueba Estadística": "Chi-Cuadrado (χ²)",
                "Estadístico": round(chi2, 4),
                "p-value (numérico)": p_val,
            }
        )

    df_resumen = pd.DataFrame(resultados)

    return df_resumen
